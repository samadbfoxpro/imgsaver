<?php
// calendar-component.php
// حرفه‌ای: تقویم میلادی با نمایش ماه‌ها و روزهای دارای عکس

if (isset($_GET['ajax_calendar'])) {
    ob_start();
    $year = isset($_GET['year']) ? intval($_GET['year']) : null;
    $month = isset($_GET['month']) ? intval($_GET['month']) : null;

    try {
        if ($year !== null && ($year < 2000 || $year > intval(date('Y')) + 1)) {
            throw new Exception('سال نامعتبر است');
        }
        renderCalendarAjax($year, $month);
        $content = ob_get_clean();
        header('Content-Type: application/json; charset=utf-8');
        echo json_encode(['html' => $content]);
    } catch (Exception $e) {
        header('Content-Type: application/json; charset=utf-8');
        echo json_encode(['html' => '<div style="color:#f44336;padding:40px;text-align:center;">خطا: ' . htmlspecialchars($e->getMessage()) . '</div>']);
    }
    exit;
}

echo '<link rel="stylesheet" href="css/calendar-modern-dark.css">';

function getImagesByDate($year, $month = null, $day = null) {
    global $all_images;
    $dates = [];
    if (!isset($all_images) || !is_array($all_images)) return $dates;

    foreach ($all_images as $img) {
        if (!isset($img['date'])) continue;
        $parts = explode('-', $img['date']);
        if (count($parts) == 3) {
            [$fYear, $fMonth, $fDay] = $parts;
            if ($fYear == $year && ($month === null || $fMonth == str_pad($month, 2, '0', STR_PAD_LEFT))) {
                if ($day === null || $fDay == str_pad($day, 2, '0', STR_PAD_LEFT)) {
                    $dates[] = $img['date'];
                }
            }
        }
    }
    return $dates;
}

function renderCalendarAjax($year = null, $month = null) {
    global $all_images;

    if (!is_array($all_images) || count($all_images) == 0) {
        echo '<div style="color:#f44336;text-align:center;padding:40px;">⚠️ داده‌ای برای نمایش تقویم وجود ندارد.</div>';
        return;
    }

    $months = [
        '01' => 'ژانویه','02' => 'فوریه','03' => 'مارس','04' => 'آوریل',
        '05' => 'مه','06' => 'ژوئن','07' => 'جولای','08' => 'اوت',
        '09' => 'سپتامبر','10' => 'اکتبر','11' => 'نوامبر','12' => 'دسامبر'
    ];

    // مرحله ۱: انتخاب سال
    if ($year === null) {
        $years = [];
        foreach ($all_images as $img) {
            $y = substr($img['date'], 0, 4);
            if (!in_array($y, $years)) $years[] = $y;
        }
        sort($years);
        echo '<h3>انتخاب سال</h3><div style="display:flex;flex-wrap:wrap;gap:10px;justify-content:center;">';
        foreach ($years as $y) {
            echo '<button class="calendar-btn has-content" onclick="selectYear(' . $y . ')">' . $y . '</button>';
        }
        echo '</div>';
    }
    // مرحله ۲: انتخاب ماه
    elseif ($month === null) {
        echo '<h3>سال ' . $year . '</h3>';
        echo '<div class="calendar-grid">';
        foreach ($months as $m => $name) {
            $hasImage = count(getImagesByDate($year, $m)) > 0;
            $cls = 'calendar-btn' . ($hasImage ? ' has-content' : ' disabled');
            echo '<button class="' . $cls . '" ' . ($hasImage ? 'onclick="selectMonth(' . $year . ',' . (int)$m . ')"' : 'disabled') . '>' . $name . '</button>';
        }
        echo '</div><button class="calendar-back" onclick="loadCalendar()">بازگشت به سال‌ها</button>';
    }
    // مرحله ۳: انتخاب روز
    else {
        $daysInMonth = cal_days_in_month(CAL_GREGORIAN, $month, $year);
        echo '<button class="calendar-back" onclick="loadCalendar(' . $year . ')">بازگشت به ماه‌ها</button>';
        echo '<h3>' . $months[str_pad($month, 2, '0', STR_PAD_LEFT)] . ' ' . $year . '</h3>';
        echo '<div class="calendar-header">';
        foreach (["ش","ی","د","س","چ","پ","ج"] as $wd) echo '<div>' . $wd . '</div>';
        echo '</div><div class="calendar-grid">';
        $firstDayOfWeek = date('w', strtotime("$year-" . str_pad($month,2,'0',STR_PAD_LEFT) . "-01"));
        for ($i = 0; $i < $firstDayOfWeek; $i++) echo '<div class="calendar-day-cell"></div>';
        for ($d = 1; $d <= $daysInMonth; $d++) {
            $hasImage = count(getImagesByDate($year, $month, $d)) > 0;
            $cls = 'calendar-btn' . ($hasImage ? ' has-content' : ' disabled');
            echo '<button class="' . $cls . '" ' . ($hasImage ? 'onclick="selectDay(' . $year . ',' . $month . ',' . $d . ')"' : 'disabled') . '>' . $d . '</button>';
        }
        echo '</div>';
    }
}

function renderCalendar($year = null, $month = null) {
    echo '<div id="calendarContainer" class="calendar-container">';
    renderCalendarAjax($year, $month);
    echo '</div>';
}

<script>
async function loadCalendar(year = null, month = null) {
    const container = document.getElementById('calendarContainer');
    container.innerHTML = '<div style="padding:40px;text-align:center;color:#aaa;">در حال بارگذاری...</div>';

    const params = new URLSearchParams();
    params.append('ajax_calendar', '1');
    if (year) params.append('year', year);
    if (month) params.append('month', month);

    try {
        const response = await fetch('calendar-component.php?' + params.toString());
        const data = await response.json();
        if (!data || !data.html) throw new Error('پاسخ نامعتبر از سرور');
        container.innerHTML = data.html;
    } catch (err) {
        console.error('Calendar load error:', err);
        container.innerHTML = '<div style="color:#f44336;text-align:center;padding:40px;">⚠️ خطا در بارگذاری تقویم<br>' + err.message + '</div>';
    }
}

function selectYear(year) {
    loadCalendar(year, null);
}
function selectMonth(year, month) {
    loadCalendar(year, month);
}
function selectDay(year, month, day) {
    const daysList = window.daysList || [];
    const target = `${year}-${String(month).padStart(2,'0')}-${String(day).padStart(2,'0')}`;
    const page = daysList.indexOf(target) + 1;
    if (page > 0) {
        window.location.search = '?page=' + page;
    } else {
        alert('تاریخ موردنظر یافت نشد!');
    }
}
</script>
