        let loadedCount = 10; // تعداد لود شده فعلی
        const totalFiles = <?= count($current_files) ?>;
        const loadMoreBtn = document.getElementById('loadMoreBtn');

        function loadMore() {
            const hiddenCards = document.getElementById('hiddenCards');
            const cards = hiddenCards.children;
            let count = 0;

            for (let i = 0; i < cards.length && count < 10; i++) {
                const card = cards[i];
                document.getElementById('galleryGrid').appendChild(card);
                loadedCount++;
                count++;
            }

            // اگر کمتر از 10 تا باقی مانده بود، دکمه را غیرفعال کن
            if (cards.length <= 0) {
                loadMoreBtn.disabled = true;
                loadMoreBtn.textContent = 'تمام موارد نمایش داده شد';
            }
        }

        function showNotification(msg) {
            const notif = document.getElementById('notif');
            notif.textContent = msg;
            notif.classList.add('show');
            setTimeout(() => notif.classList.remove('show'), 1700);
        }
        function copyToClipboard(btn) {
            const text = btn.closest('.copy-box').dataset.copy;
            navigator.clipboard.writeText(text).then(() => showNotification('کپی شد!'));
        }
        function openModal(src) {
            document.getElementById('modalImg').src = src;
            document.getElementById('imageModal').style.display = 'block';
        }
        function closeModal() {
            document.getElementById('imageModal').style.display = 'none';
